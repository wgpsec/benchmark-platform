# Web UI Authentication Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add cookie-based authentication to the Web UI so only admin (default team token) gets full access, team observers get a read-only scoreboard, and unauthenticated users see a login page.

**Architecture:** Starlette middleware intercepts all `/web/*` requests, validates a signed cookie (via `itsdangerous`), and injects user context into `request.state`. Login page accepts an Agent-Token, validates it against the teams table, and issues a signed cookie. Observer role is restricted to a standalone scoreboard page.

**Tech Stack:** FastAPI/Starlette middleware, itsdangerous (URLSafeTimedSerializer), Jinja2, Tailwind CSS, Alpine.js

### Task 2: Create Auth Middleware

**Files:**
- Create: `benchmark_platform/web/auth_middleware.py`

- [ ] **Step 1: Create the auth middleware file**

```python
"""Cookie-based authentication middleware for the Web UI."""

import os
import secrets
from typing import Optional

from itsdangerous import URLSafeTimedSerializer, BadSignature, SignatureExpired
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
from starlette.responses import RedirectResponse

_MAX_AGE = 7 * 24 * 3600  # 7 days in seconds
_COOKIE_NAME = "bp_session"

# Paths that bypass auth entirely
_PUBLIC_PREFIXES = ("/api/", "/mcp", "/static/", "/vnc/")
_PUBLIC_PATHS = ("/web/login", "/web/login/")


def get_secret_key() -> str:
    key = os.environ.get("SECRET_KEY")
    if not key:
        key = secrets.token_hex(32)
        import logging
        logging.getLogger("benchmark_platform").warning(
            "SECRET_KEY not set — generated ephemeral key. Sessions will not survive restart."
        )
    return key


_secret_key: Optional[str] = None


def _get_serializer() -> URLSafeTimedSerializer:
    global _secret_key
    if _secret_key is None:
        _secret_key = get_secret_key()
    return URLSafeTimedSerializer(_secret_key, salt="bp-session")


def create_session_cookie(team_id: str, role: str, team_name: str) -> str:
    s = _get_serializer()
    return s.dumps({"team_id": team_id, "role": role, "team_name": team_name})


def verify_session_cookie(cookie_value: str) -> Optional[dict]:
    s = _get_serializer()
    try:
        data = s.loads(cookie_value, max_age=_MAX_AGE)
        return data
    except (BadSignature, SignatureExpired):
        return None


class AuthMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        path = request.url.path

        # Skip auth for public paths
        if any(path.startswith(p) for p in _PUBLIC_PREFIXES):
            return await call_next(request)
        if path in _PUBLIC_PATHS:
            return await call_next(request)
        # Root redirect doesn't need auth (it redirects to /web/dashboard which will trigger auth)
        if path == "/":
            return await call_next(request)

        # Check cookie
        cookie = request.cookies.get(_COOKIE_NAME)
        user = verify_session_cookie(cookie) if cookie else None

        if user is None:
            return RedirectResponse("/web/login", status_code=302)

        # Observer can only access scoreboard
        if user["role"] == "observer" and not path.startswith("/web/scoreboard"):
            return RedirectResponse("/web/scoreboard", status_code=302)

        # Inject user into request state
        request.state.user = user
        return await call_next(request)
```

```python
@web_router.post("/login")
async def login_submit(request: Request):
    form = await request.form()
    token = form.get("token", "").strip()

    if not token:
        return templates.TemplateResponse(
            request, "pages/login.html", context={"error": "Please enter a token"}
        )

    team = get_team_by_token(token)
    if team is None:
        return templates.TemplateResponse(
            request, "pages/login.html", context={"error": "Invalid token"}
        )

    default_team = get_or_create_default_team()
    role = "admin" if team["id"] == default_team["id"] else "observer"
    cookie_value = create_session_cookie(team["id"], role, team["name"])

    redirect_url = "/web/dashboard" if role == "admin" else "/web/scoreboard"
    response = RedirectResponse(redirect_url, status_code=302)
    response.set_cookie(
        _COOKIE_NAME,
        cookie_value,
        max_age=7 * 24 * 3600,
        httponly=True,
        samesite="lax",
    )
    return response
```

- [ ] **Step 1: Return 401 for HTMX requests instead of redirect**

In the `AuthMiddleware.dispatch` method, before the redirect responses, add handling for HTMX requests that should get a 401 (so HTMX can handle it client-side) instead of a 302 redirect (which HTMX would follow silently):

Replace the `if user is None:` block with:

```python
        if user is None:
            # HTMX requests get 401 so the client can redirect via JS
            if request.headers.get("HX-Request"):
                from starlette.responses import Response
                return Response(
                    status_code=401,
                    headers={"HX-Redirect": "/web/login"},
                )
            return RedirectResponse("/web/login", status_code=302)
```
